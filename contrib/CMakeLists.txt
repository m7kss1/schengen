set_property(DIRECTORY PROPERTY EXCLUDE_FROM_ALL 1)

function(add_contrib cmake_folder)
  if (ARGN)
    set(base_folders ${ARGN})
  else()
    set(base_folders ${cmake_folder})
  endif()

  foreach (base_folder IN LISTS base_folders)
    if (NOT IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${base_folder}")
      message(
        FATAL_ERROR
          "No such base folder '${base_folder}' (for '${cmake_folder}' cmake folder)."
      )
    endif()

    file(GLOB contrib_files "${base_folder}/*")
    if (NOT contrib_files)
      message(STATUS "submodule ${base_folder} is missing or empty. to fix run:")
      message(STATUS "    git submodule update --init --recursive")
      return()
    endif()
  endforeach()

  message(STATUS "Adding contrib module ${base_folders} (configuring with ${cmake_folder})")
  add_subdirectory(${cmake_folder})
endfunction()

if (ENABLE_VORTEX)
  set(_enable_vortex_stack ON)
  if (NOT IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/corrosion")
    message(WARNING "Corrosion is missing; skipping Vortex")
    set(_enable_vortex_stack OFF)
  else ()
    file(GLOB _corrosion_files "${CMAKE_CURRENT_SOURCE_DIR}/corrosion/*")
    if (NOT _corrosion_files)
      message(WARNING "Corrosion is empty; skipping Vortex")
      set(_enable_vortex_stack OFF)
    endif ()
  endif ()

  if (_enable_vortex_stack)
    add_contrib(corrosion-cmake corrosion)
    add_contrib(vortex-cmake vortex)
  endif ()
else ()
  message(STATUS "Skipping contrib module vortex (ENABLE_VORTEX=OFF)")
endif ()
